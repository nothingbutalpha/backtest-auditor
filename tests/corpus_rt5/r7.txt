#### F01_add_prefix
import pandas as pd
from sklearn.ensemble import GradientBoostingRegressor
df = pd.read_csv("px.csv", index_col=0, parse_dates=True)
fwd = df["close"].shift(-5) / df["close"] - 1
data = pd.DataFrame({"mom": df["close"].pct_change(20), "fwd": fwd}).dropna()
data = data.add_prefix("f_")
X = data[["f_mom", "f_fwd"]]
model = GradientBoostingRegressor().fit(X.iloc[:1000], fwd.loc[X.index].iloc[:1000])
