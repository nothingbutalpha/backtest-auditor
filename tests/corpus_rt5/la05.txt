import pandas as pd
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import Ridge
df = pd.read_csv("feat.csv", parse_dates=["date"], index_col="date")
X = df[["a", "b", "c"]]
y = df["close"].pct_change().shift(-1)
Xs = StandardScaler().fit_transform(X)
X_train, X_test = Xs[:2000], Xs[2000:]
model = Ridge().fit(X_train, y[:2000])
print(model.score(X_test, y[2000:len(Xs)]))
