#### S01_sort_index_axis1_then_iloc
import pandas as pd
from sklearn.ensemble import GradientBoostingRegressor
df = pd.read_csv("px.csv", index_col=0, parse_dates=True)
fwd = df["close"].shift(-5) / df["close"] - 1
data = pd.concat([df["close"].pct_change(20), df["close"].pct_change(5), fwd], axis=1, keys=["mom", "rev", "zfwd"]).dropna()
data = data.sort_index(axis=1)
X = data.iloc[:, :-1]
model = GradientBoostingRegressor().fit(X.iloc[:1000], fwd.loc[X.index].iloc[:1000])

#### T01_long_mask_select
import pandas as pd, numpy as np
df = pd.read_parquet("panel.parquet")
df["fwd_ret"] = df.groupby("ticker")["close"].pct_change().shift(-1)
mkt = df.groupby("date")["fwd_ret"].mean()
bench = df.groupby("date")["close"].mean().pct_change().shift(-1)
long = mkt > 0
pnl = bench[long].sum() - bench[~long].sum()
#### T02_short_rank_loc

strat = bench.copy()
strat.loc[short] = -bench.loc[short]

#### U02_clean_long_leg_ic
import pandas as pd, numpy as np
df = pd.read_parquet("panel.parquet")
df["fwd_ret"] = df.groupby("ticker")["close"].pct_change().shift(-1)
mkt = df.groupby("date")["fwd_ret"].mean()
bench = df.groupby("date")["close"].mean().pct_change().shift(-1)
df["q"] = df.groupby("date")["mom"].transform(lambda s: pd.qcut(s, 5, labels=False))
long = df[df["q"] == 4].groupby("date")["fwd_ret"].mean()
short = df[df["q"] == 0].groupby("date")["fwd_ret"].mean()
trade = long - short
print(trade.mean(), np.sign(trade).mean())
