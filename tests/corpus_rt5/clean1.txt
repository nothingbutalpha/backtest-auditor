#### C10_rolling_sharpe_lagged
import pandas as pd, numpy as np
df = pd.read_csv("px.csv", index_col=0, parse_dates=True)
df["fwd_ret"] = df["close"].pct_change().shift(-1)
df["signal"] = np.sign(df["close"] - df["close"].rolling(50).mean())
edge = df["signal"] * df["fwd_ret"]
weight = (edge.rolling(60).mean() / edge.rolling(60).std()).shift(1)
df["strat"] = weight * df["signal"] * df["fwd_ret"]
#### C11_lgb_with_label_and_valid
import pandas as pd, lightgbm as lgb
df = pd.read_csv("px.csv", index_col=0, parse_dates=True)
df["mom"] = df["close"].pct_change(20)
df["vol"] = df["close"].pct_change().rolling(20).std()
lab = (df["close"].shift(-1) > df["close"]).astype(int)
tr, va = slice(0, 2000), slice(2000, None)
dtr = lgb.Dataset(df[["mom", "vol"]].iloc[tr], lab.iloc[tr])
dva = lgb.Dataset(df[["mom", "vol"]].iloc[va], lab.iloc[va], reference=dtr)
bst = lgb.train({"objective": "binary"}, dtr, valid_sets=[dva])
