#### N04_fill_return_feeds_size
import pandas as pd, numpy as np
df = pd.read_csv("px.csv", index_col=0, parse_dates=True)
entry_px = df["open"].shift(-1)
exit_px = df["open"].shift(-2)
df["signal"] = np.sign(df["close"].pct_change(10))
trade = df["signal"] * (exit_px / entry_px - 1)
df["size"] = trade.rolling(50).mean() / trade.rolling(50).std()
df["pnl"] = df["size"] * df["signal"] * (exit_px / entry_px - 1)
#### N05_ic_weighting_cross_sectional
import pandas as pd
df = pd.read_parquet("panel.parquet")
df["fwd_ret"] = df.groupby("ticker")["close"].pct_change().shift(-1)
ic = df.groupby("date").apply(lambda g: g["mom"].corr(g["fwd_ret"]))
w = ic.rolling(12).mean()
df["signal"] = df["mom"] * df["date"].map(w)
