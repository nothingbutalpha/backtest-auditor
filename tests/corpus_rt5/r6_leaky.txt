df = pd.read_parquet("panel.parquet")
df["fwd_ret"] = df.groupby("ticker")["close"].pct_change().shift(-1)
mkt = df.groupby("date")["fwd_ret"].mean()

pnl = np.where(long_short > 0, bench, -bench)

#### A06_ls_helper_backtest
import pandas as pd, numpy as np
def backtest(w, r):
    return (w * r).sum(axis=1)
df = pd.read_parquet("panel.parquet")
df["fwd_ret"] = df.groupby("ticker")["close"].pct_change().shift(-1)
fr = df.pivot(index="date", columns="ticker", values="fwd_ret")
long_short = np.sign(fr.sub(fr.mean(axis=1), axis=0))
pnl = backtest(long_short, fr)
