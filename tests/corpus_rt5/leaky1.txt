#### L18_ic_used_to_choose_feature
import pandas as pd
df = pd.read_parquet("panel.parquet")
df["fwd_ret"] = df.groupby("ticker")["close"].pct_change().shift(-1)
feats = ["mom", "rev", "val"]
ic = {f: df[f].corr(df["fwd_ret"]) for f in feats}
best = max(ic, key=ic.get)
df["signal"] = df.groupby("date")[best].rank(pct=True)
df["pnl"] = (df["signal"] > 0.8) * df["fwd_ret"]
