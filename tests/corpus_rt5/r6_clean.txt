import pandas as pd
from sklearn.ensemble import GradientBoostingRegressor
df = pd.read_csv("px.csv", index_col=0, parse_dates=True)
fwd = df["close"].shift(-5) / df["close"] - 1
data = pd.concat([df["close"].pct_change(20), df["close"].pct_change(5), fwd], axis=1).dropna()
X, y = data.iloc[:, :-1], data.iloc[:, -1]
split = int(len(data) * 0.7)
model = GradientBoostingRegressor().fit(X.iloc[:split], y.iloc[:split])
print(model.score(X.iloc[split:], y.iloc[split:]))

#### K12_symbolic_same_param
import pandas as pd
def build(df, horizon=5):
    df["fwd_ret"] = df["close"].shift(-horizon) / df["close"] - 1
    df["ret_h"] = df["close"] / df["close"].shift(horizon) - 1
    df["prev_fwd"] = df["fwd_ret"].shift(horizon)
    df["signal"] = (df["prev_fwd"] > 0) & (df["ret_h"] > 0)
    return df
