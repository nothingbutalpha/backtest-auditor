#### P01_neutral_tilt_from_groupby
import pandas as pd, numpy as np
df = pd.read_parquet("panel.parquet")
df["fwd_ret"] = df.groupby("ticker")["close"].pct_change().shift(-1)
mkt = df.groupby("date")["fwd_ret"].mean()
tilt = np.sign(mkt)
bench = df.groupby("date")["ret"].mean()
pnl = tilt * bench
#### P02_booked_then_neutral_rolling
import pandas as pd, numpy as np
df = pd.read_csv("px.csv", index_col=0, parse_dates=True)
df["fwd_ret"] = df["close"].pct_change().shift(-1)
df["signal"] = np.sign(df["close"].diff(20))
r = df["signal"] * df["fwd_ret"]
k = r.rolling(20).mean().clip(0, None) * 10
df["pnl"] = k * df["signal"] * df["fwd_ret"]
#### P03_loc_mask_outcome_into_model
import pandas as pd
from sklearn.linear_model import Ridge
df = pd.read_parquet("panel.parquet")
df["fwd_ret"] = df.groupby("ticker")["close"].pct_change().shift(-5)
liquid = df["volume"] > 1e6
df["liq_fwd"] = df.loc[liquid, "fwd_ret"]
X = df[["mom", "liq_fwd"]].fillna(0)
model = Ridge().fit(X.iloc[:5000], df["fwd_ret"].fillna(0).iloc[:5000])
#### P04_drop_by_position_iloc
import pandas as pd
from sklearn.ensemble import RandomForestRegressor
df = pd.read_csv("px.csv", index_col=0, parse_dates=True)
df["mom"] = df["close"].pct_change(20)
df["fwd_ret"] = df["close"].shift(-5) / df["close"] - 1
df["target"] = (df["fwd_ret"] > 0).astype(int)
df = df.dropna()
X = df.drop(columns=["target"])
model = RandomForestRegressor().fit(X.iloc[:2000], df["target"].iloc[:2000])

#### P10_dataframe_then_signal
import pandas as pd, numpy as np
df = pd.read_csv("px.csv", index_col=0, parse_dates=True)
fwd = df["close"].shift(-5) / df["close"] - 1
tbl = pd.DataFrame({"mom": df["close"].pct_change(20), "fwd": fwd})
tbl["position"] = np.sign(tbl["fwd"])
tbl["pnl"] = tbl["position"] * df["close"].pct_change()
