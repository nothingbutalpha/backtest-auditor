import pandas as pd
def triple_barrier_labels(close, horizon=10):
    labels = pd.Series(0, index=close.index, dtype=int)
    for i in range(len(close) - horizon):
        if close.iloc[i + 1: i + 1 + horizon].max() > close.iloc[i] * 1.03:
            labels.iloc[i] = 1
    return labels
px = pd.read_csv("spy.csv", index_col="date", parse_dates=True)
y = triple_barrier_labels(px["close"])
px.loc[:, "lab"] = y
px["position"] = px["lab"].shift(1)
px["pnl"] = px["position"] * px["close"].pct_change()

#### G9_fill_low_next_into_exit_price
import pandas as pd
df = pd.read_csv("spy.csv", index_col="date", parse_dates=True)
want_buy = df["close"] < df["close"].rolling(20).mean()
limit_px = df["close"] * 0.99
filled = want_buy & (df["low"].shift(-1) <= limit_px)
best = df[["low", "high"]].shift(-1)
trade_ret = (best["high"] / limit_px - 1)[filled]
print(trade_ret.mean())

#### G12_fill_mask_cumsum_count_signal
import pandas as pd
df = pd.read_csv("spy.csv", index_col="date", parse_dates=True)
want_buy = df["close"] < df["close"].rolling(20).mean()
limit_px = df["close"] * 0.99
filled = want_buy & (df["low"].shift(-1) <= limit_px)
streak = filled.cumsum()
df["position"] = (streak.diff(5) > 1).astype(int)
df["pnl"] = df["position"].shift(1) * df["close"].pct_change()
#### G13_fill_mask_via_numpy
import numpy as np, pandas as pd
d = pd.read_csv("spy.csv")
c, lo = d["close"].to_numpy(), d["low"].to_numpy()
lim = c * 0.99
want = c < pd.Series(c).rolling(20).mean().to_numpy()
filled = np.zeros(len(c), dtype=bool)
filled[:-1] = want[:-1] & (lo[1:] <= lim[:-1])
size = np.zeros(len(c))
size[:-1] = (lim[:-1] - lo[1:]) / lim[:-1]
r = np.zeros(len(c))
r[:-5] = c[5:] / lim[:-5] - 1
print((size * r)[filled].mean())
