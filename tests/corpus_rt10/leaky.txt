import pandas as pd
def triple_barrier_labels(close, horizon=10):
    labels = pd.Series(0, index=close.index, dtype=int)
    for i in range(len(close) - horizon):
        path = close.iloc[i + 1: i + 1 + horizon]
        if path.max() > close.iloc[i] * 1.03:
            labels.iloc[i] = 1
    return labels
px = pd.read_csv("spy.csv", index_col="date", parse_dates=True)
close = px["close"]
y = triple_barrier_labels(close)
trade_days = y[y == 1].index
ret = close.pct_change().shift(-1)
strat = ret.loc[trade_days]
print(strat.mean() / strat.std() * 252 ** 0.5)

def make_labels(close, h=5):
