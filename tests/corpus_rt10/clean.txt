#### N08_labels_then_fit_target_col
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
def triple_barrier_labels(close, horizon=10):
    labels = pd.Series(0, index=close.index, dtype=int)
    for i in range(len(close) - horizon):
        path = close.iloc[i + 1: i + 1 + horizon]
        if path.max() > close.iloc[i] * 1.03:
            labels.iloc[i] = 1
    return labels
px = pd.read_csv("spy.csv", index_col="date", parse_dates=True)
px["target"] = triple_barrier_labels(px["close"])
px["mom"] = px["close"].pct_change(20)
X = px[["mom"]].iloc[20:-10]
RandomForestClassifier().fit(X.iloc[:1000], px["target"].iloc[20:1020])

#### N13_pivot_fwd_decile_eval
import pandas as pd
df = pd.read_parquet("panel.parquet")
df["fwd_ret"] = df.groupby("ticker")["close"].pct_change().shift(-1)
df["dec"] = df.groupby("date")["mom"].transform(lambda s: pd.qcut(s, 10, labels=False))
tbl = df.pivot_table(index="date", columns="dec", values="fwd_ret", aggfunc="mean")
ls = tbl[9] - tbl[0]
print(ls.mean() / ls.std() * 252 ** 0.5)
#### N14_apply_rank_cross_section
