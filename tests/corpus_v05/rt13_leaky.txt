#### X01_idx_via_dict
import numpy as np, pandas as pd
bars = pd.read_parquet("spy.parquet")
o, lo, c = bars["open"].to_numpy(), bars["low"].to_numpy(), bars["close"].to_numpy()
sigs = {"long": np.flatnonzero(bars["close"].shift(-5).to_numpy() > c)}
trades = []
for i in sigs["long"]:
    if i + 6 >= len(c):
        break
    limit = c[i] * 0.995
    if lo[i + 1] <= limit:
        trades.append(c[i + 6] / min(limit, o[i + 1]) - 1)
print(np.mean(trades))

bars = pd.read_parquet("spy.parquet")

#### X03_idx_via_generator
import numpy as np, pandas as pd
bars = pd.read_parquet("spy.parquet")
o, lo, c = bars["open"].to_numpy(), bars["low"].to_numpy(), bars["close"].to_numpy()
def signal_days():
    for k in range(len(c) - 5):
        if c[k + 5] > c[k]:
            yield k
trades = []
for i in signal_days():
    if i + 6 >= len(c):
        break
    limit = c[i] * 0.995
    if lo[i + 1] <= limit:
        trades.append(c[i + 6] / min(limit, o[i + 1]) - 1)
print(np.mean(trades))
#### X04_idx_from_label_series_index
import numpy as np, pandas as pd
bars = pd.read_parquet("spy.parquet")
o, lo, c = bars["open"].to_numpy(), bars["low"].to_numpy(), bars["close"].to_numpy()
fwd = bars["close"].pct_change(5).shift(-5).reset_index(drop=True)
idx = fwd[fwd > 0.01].index
trades = []
for i in idx:

        break
    limit = c[i] * 0.995
    if lo[i + 1] <= limit:
        trades.append(c[i + 6] / min(limit, o[i + 1]) - 1)

#### X05_fill_laundered_via_list_then_condition
import numpy as np, pandas as pd
d = pd.read_csv("spy.csv")
o, lo, c = (d[k].to_numpy() for k in ("open", "low", "close"))
sma = d["close"].rolling(20).mean().to_numpy()
HOLD = 5
trades = []
for i in range(20, len(c) - HOLD - 1):
    if c[i] < sma[i] and (not trades or trades[-1] > 0):
        limit = c[i] * 0.99
        if lo[i + 1] <= limit:
            fill = min(limit, o[i + 1])
            trades.append(c[i + 1 + HOLD] / fill - 1)
print(np.mean(trades))
#### X06_fill_laundered_dict_state
import numpy as np, pandas as pd
d = pd.read_csv("spy.csv")
o, lo, c = (d[k].to_numpy() for k in ("open", "low", "close"))
sma = d["close"].rolling(20).mean().to_numpy()
HOLD = 5
state = {"last": 0.0}
log = []
for i in range(20, len(c) - HOLD - 1):
    if c[i] < sma[i] and state["last"] >= 0:
        limit = c[i] * 0.99
        if lo[i + 1] <= limit:
            fill = min(limit, o[i + 1])
            r = c[i + 1 + HOLD] / fill - 1
            log.append(r)
            state["last"] = r
print(np.mean(log))
