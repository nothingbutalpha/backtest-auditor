#### R01_signal_idx_from_future_indirect
import numpy as np, pandas as pd
def simulate(bars, signal_idx, hold=5):
    o, lo, c = bars["open"].to_numpy(), bars["low"].to_numpy(), bars["close"].to_numpy()
    trades = []
    for i in signal_idx:
        if i + 1 + hold >= len(c):
            break
        limit = c[i] * 0.995
        if lo[i + 1] <= limit:
            fill = min(limit, o[i + 1])
            trades.append(c[i + 1 + hold] / fill - 1)
    return trades
bars = pd.read_parquet("spy.parquet")
up_next = bars["close"].shift(-5) > bars["close"]
mask = up_next.to_numpy()
idx = np.flatnonzero(mask)
print(np.mean(simulate(bars, idx)))

#### T01_lambda_negative_shift
import pandas as pd
panel = pd.read_parquet("crypto.parquet").sort_values(["symbol", "date"])
panel["ret"] = panel.groupby("symbol")["close"].pct_change()
panel["mom"] = panel.groupby("symbol")["close"].transform(lambda s: s.pct_change(7).shift(-1))
