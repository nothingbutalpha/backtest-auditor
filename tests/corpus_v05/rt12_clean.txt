#### K01_stop_entry_loop_breakout
import numpy as np, pandas as pd
d = pd.read_csv("spy.csv")
o, h, lo, c = (d[k].to_numpy() for k in ("open", "high", "low", "close"))
hh = d["high"].rolling(20).max().to_numpy()
out = []
for i in range(20, len(c) - 4):
    stop_px = hh[i]
    if h[i + 1] >= stop_px:
        fill = max(stop_px, o[i + 1])
        out.append(c[i + 4] / fill - 1 - 0.001)
print(np.mean(out))
#### K02_trailing_stop_sim_while
import numpy as np, pandas as pd
d = pd.read_csv("spy.csv")
o, h, lo, c = (d[k].to_numpy() for k in ("open", "high", "low", "close"))
sma = d["close"].rolling(50).mean().to_numpy()
rets, i = [], 50
while i < len(c) - 1:
    if c[i] > sma[i]:
        entry = o[i + 1]
        peak = entry
        j = i + 1
        while j < len(c) - 1:
            peak = max(peak, c[j])
            stop = peak * 0.95
            if lo[j + 1] <= stop:
                rets.append(min(stop, o[j + 1]) / entry - 1)
                break
            j += 1
        i = j + 1
    else:
        i += 1
print(np.mean(rets))
#### K03_signal_idx_loop_next_open
import numpy as np, pandas as pd
bars = pd.read_parquet("spy.parquet")
o, c = bars["open"].to_numpy(), bars["close"].to_numpy()
rsi_like = bars["close"].diff().rolling(14).mean().to_numpy()
idx = np.flatnonzero(rsi_like < -1.0)
trades = []
for i in idx:
    if i + 6 >= len(c):
        break
    trades.append(c[i + 6] / o[i + 1] - 1 - 0.0005)
print(np.mean(trades))
#### K04_cooldown_from_closed_trade
import numpy as np, pandas as pd
d = pd.read_csv("spy.csv")
o, lo, c = (d[k].to_numpy() for k in ("open", "low", "close"))
sma = d["close"].rolling(20).mean().to_numpy()
HOLD = 5
trades, busy_until = [], -1
for i in range(20, len(c) - HOLD - 1):
    if i <= busy_until:
        continue
    if c[i] < sma[i]:
        limit = c[i] * 0.99
        if lo[i + 1] <= limit:
            fill = min(limit, o[i + 1])
            trades.append(c[i + 1 + HOLD] / fill - 1)
            busy_until = i + 1 + HOLD
print(np.mean(trades))
#### K05_numpy_labels_into_fit
import numpy as np
from sklearn.linear_model import LogisticRegression
p = np.load("closes.npy")
H = 10
labels = np.zeros(len(p) - H)
for i in range(len(p) - H):
    labels[i] = 1.0 if p[i + H] > p[i] else 0.0
X = np.column_stack([np.r_[np.nan, np.diff(np.log(p))][:len(labels)]])
X[0] = 0
LogisticRegression().fit(X[:1000], labels[:1000])
print(np.bincount(labels.astype(int)))
#### K06_transform_lambda_rolling_shift_z
import pandas as pd
panel = pd.read_parquet("crypto.parquet").sort_values(["symbol", "date"])
panel["ret"] = panel.groupby("symbol")["close"].pct_change()
panel["vol"] = panel.groupby("symbol")["ret"].transform(lambda s: s.rolling(20).std().shift(1))
panel["w"] = 0.01 / panel["vol"]
pnl = (panel["w"] * panel["ret"]).groupby(panel["date"]).sum()
#### K07_apply_helper_ewm_lag
import pandas as pd
def ewm_sig(s, span=10):
    return (s.ewm(span=span).mean() - s.ewm(span=4 * span).mean()).shift(1)
panel = pd.read_parquet("crypto.parquet").sort_values(["symbol", "date"])
panel["ret"] = panel.groupby("symbol")["close"].pct_change()
panel["sig"] = panel.groupby("symbol", group_keys=False)["close"].apply(ewm_sig)
panel["position"] = (panel["sig"] > 0).astype(float)
pnl = (panel["position"] * panel["ret"]).groupby(panel["date"]).mean()
#### K08_pipe_lagged
import pandas as pd
def add_sig(df):
    df = df.copy()
    df["sig"] = df.groupby("symbol")["close"].transform(lambda s: s.pct_change(5).shift(1))
    return df
panel = pd.read_parquet("crypto.parquet").sort_values(["symbol", "date"])
panel["ret"] = panel.groupby("symbol")["close"].pct_change()
panel = panel.pipe(add_sig)
panel["position"] = (panel["sig"] > 0).astype(float)
pnl = (panel["position"] * panel["ret"]).groupby(panel["date"]).mean()
#### K09_bracket_order_iterrows_next
import pandas as pd
bars = pd.read_csv("es_5m.csv", index_col=0, parse_dates=True)
bars["ma"] = bars["close"].rolling(12).mean()
res = []
rows = list(bars.itertuples())
for k in range(12, len(rows) - 1):
    cur, nxt = rows[k], rows[k + 1]
    if cur.close > cur.ma:
        entry = nxt.open
        if nxt.high >= entry + 2.0:
            res.append(2.0)
        elif nxt.low <= entry - 1.0:
            res.append(-1.0)
        else:
            res.append(nxt.close - entry)
print(sum(res))
