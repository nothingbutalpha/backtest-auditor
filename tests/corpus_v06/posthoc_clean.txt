#### live_ols_hedge
import numpy as np, pandas as pd, statsmodels.api as sm
px = pd.read_csv("pair.csv", index_col=0, parse_dates=True).iloc[-500:]
a, b = np.log(px["A"]), np.log(px["B"])
hedge = sm.OLS(a, sm.add_constant(b)).fit().params["B"]
spread = a - hedge * b
z = (spread.iloc[-1] - spread.mean()) / spread.std()
signal = -np.sign(z) if abs(z) > 2 else 0
print("today's signal", signal)

#### groupby_ticker_rank_lagged_rolling
import pandas as pd
panel = pd.read_parquet("p.parquet").sort_values(["ticker", "date"])
panel["mom"] = panel.groupby("ticker")["close"].pct_change(20).shift(1)
panel["signal"] = panel.groupby("date")["mom"].rank(pct=True)

#### daily_first_transform
import pandas as pd
m = pd.read_parquet("m.parquet")
m["day_open"] = m.groupby(m.index.normalize())["open"].transform("first")
m["signal"] = (m["close"] > m["day_open"]).astype(int)
print((m["signal"].shift(1) * m["close"].pct_change()).sum())

#### prev_day_stats_merged
import pandas as pd
m = pd.read_parquet("m.parquet")
daily = m["close"].resample("1D").last().shift(1)
m["prev_close"] = daily.reindex(m.index.normalize()).to_numpy()
m["signal"] = (m["close"] > m["prev_close"]).astype(int)
print((m["signal"].shift(1) * m["close"].pct_change()).sum())

#### fill_at_open_entry
import pandas as pd
bars = pd.read_csv("b.csv")
out = []
for i in range(20, len(bars) - 5):
    if bars["close"].iloc[i] < bars["close"].iloc[i - 20:i].mean():
        entry_px = bars["open"].iloc[i + 1]
        out.append(bars["close"].iloc[i + 5] / entry_px - 1)
print(sum(out))

#### pca_report_with_pnl_elsewhere
import numpy as np, pandas as pd
from sklearn.decomposition import PCA
R = pd.read_parquet("r.parquet").fillna(0)
pca = PCA(3).fit(R.to_numpy())
print(pca.explained_variance_ratio_)
w = R.rolling(20).mean().shift(1)
pnl = (np.sign(w) * R).sum(axis=1)
print(pnl.mean())

#### tss_gap_label1
import pandas as pd
from sklearn.model_selection import TimeSeriesSplit
from sklearn.linear_model import Ridge
df = pd.read_parquet("f.parquet")
df["y"] = df["close"].pct_change().shift(-1)
df = df.dropna()
X = df.drop(columns=["y", "close"])
for tr, te in TimeSeriesSplit(5).split(X):
    print(Ridge().fit(X.iloc[tr], df["y"].iloc[tr]).score(X.iloc[te], df["y"].iloc[te]))
